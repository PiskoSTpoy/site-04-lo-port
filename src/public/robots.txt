# Сайт открыт целиком: закрывать нечего — ни поиска, ни черновиков,
# ни служебных разделов.
User-agent: *
Allow: /

# Дальше — те же самые разрешения, продублированные поимённо для поисковых
# ассистентов. Прямо сейчас эти блоки не меняют ничего: всех перечисленных
# роботов уже пускает блок «*» выше. Смысл в другом. Как только в общий блок
# однажды попадёт Disallow — закрыть черновик, служебную копию, что угодно, —
# канал ответов ассистентов отключится вместе с ним и молча: обычная выдача
# при этом не просядет, и заметить потерю будет не по чему. Именной блок
# агента такую правку переживает.

# Яндекс. Два написания одного канала — нейроответы и Алиса. Нужны оба имени:
# оставить одно значит открыть канал наполовину.
User-agent: YandexAdditional
Allow: /

User-agent: YandexAdditionalBot
Allow: /

# Anthropic. Именно Claude-SearchBot — поисковый обход, а не сбор обучающих
# данных: это разные роботы с разными именами.
User-agent: Claude-SearchBot
Allow: /

User-agent: PerplexityBot
Allow: /

# OpenAI. OAI-SearchBot отвечает за показ страницы в ответах ChatGPT.
User-agent: OAI-SearchBot
